任务 3：传统边缘锐化
Roberts 使用 2×2 交叉差分核 Gx=[[1,0],[0,-1]]、Gy=[[0,1],[-1,0]]；Sobel 的水平
核为[[-1,0,1],[-2,0,2],[-1,0,1]]，垂直核为其转置方向；Prewitt 将中间行/列权重改
为 1。边缘幅值 E=√(Gx²+Gy²)。代码以 E−mean3×3(E) 作为局部细节掩膜，再计算
clip(I+0.2·D,0,255)，同时输出共同比例的边缘响应。
Laplacian 使用中心为正的四邻域核 L=[[0,-1,0],[-1,4,-1],[0,-1,0]]。亮点中心的响应
为正、相邻暗位置为负；因此锐化方向为 I+L。若采用中心为负的相反符号核，则应使用
I−L。
图 3 Roberts、Sobel、Prewitt、Laplacian 响应（各响应采用同一稳健上限）及 Laplacian 叠加结果




四种算子都能突出眼部、发丝、雀斑和轮廓。Sobel 对中心邻域加权，边缘更连续但响应也
更强；Prewitt 权重均匀；Roberts 核小，定位细而对噪声更敏感。边缘锐化只是将局部高
频掩膜叠加回原图，幅度过大会放大噪声。
Laplacian 负值处理比较
本实验对同一有符号响应 L 分别采用 |L|、线性拉伸 255·(L−Lmin)/(Lmax−Lmin)，以及
clip(I+L,0,255)。绝对值保留边缘强弱但丢失正负方向；线性拉伸能同时显示两种极性，
却把零响应映射到中间灰度，结果是边缘图而非保持原亮度的增强图；原图叠加保留低频外
观，并用正负响应提升亮暗边界对比。
             图 4 Laplacian 绝对值、线性拉伸和原图叠加截断对比




本例中原图叠加最符合增强细节的目标，因为它保留原图结构并叠加带符号的高频变化。截
断到[0,255]会在极亮或极暗处损失少量响应；绝对值法和线性拉伸会改变亮度语义，不适
合作为原图增强结果直接替代。

"""Hand-written grayscale histogram operations for uint8 images."""



from typing import Union



import numpy as np




ArrayLike = Union[np.ndarray, list]




def _gray_u8(image: ArrayLike) -> np.ndarray:

   array = np.asarray(image)

   if array.ndim != 2:

       raise ValueError("Expected a two-dimensional grayscale image")

   if not np.isfinite(array).all():

       raise ValueError("Image values must be finite")

   return np.clip(np.rint(array), 0, 255).astype(np.uint8)




def gray_histogram(image: ArrayLike) -> np.ndarray:

   """Count the occurrences of gray levels 0..255 without image libraries."""

   pixels = _gray_u8(image).ravel()

   return np.bincount(pixels, minlength=256).astype(np.int64, copy=False)




def equalize_histogram(image: ArrayLike) -> np.ndarray:

   """Apply discrete CDF equalization, mapping the first occupied CDF to zero."""

   source = _gray_u8(image)

   counts = gray_histogram(source)

   occupied = np.flatnonzero(counts)

   if occupied.size <= 1:

       return source.copy()



   cdf = counts.cumsum(dtype=np.float64) / float(source.size)

def box_filter(image: np.ndarray, size: int) -> np.ndarray:

   """Return local box means using an integral image and reflected borders."""

   source = _float_gray(image)

   size = _check_odd_size(size)

   radius = size // 2

   padded = np.pad(source, ((radius, radius), (radius, radius)), mode=_pad_mode(source))

   integral = np.pad(padded.cumsum(axis=0).cumsum(axis=1), ((1, 0), (1, 0)))

   height, width = source.shape

   bottom_right = integral[size : size + height, size : size + width]

   top_right = integral[:height, size : size + width]

   bottom_left = integral[size : size + height, :width]

   top_left = integral[:height, :width]

   return (bottom_right - top_right - bottom_left + top_left) / float(size * size)

   guide: np.ndarray,

   source: np.ndarray,

   radius: int = 4,

   epsilon: float = 650.25,

) -> np.ndarray:

   source_image = _gray(source)

       raise ValueError("guide and source must have equal shapes")
